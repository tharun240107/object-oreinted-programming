# ==========================================================
# FORWARD ALGORITHM FOR HMM (FROM SCRATCH)
# ==========================================================

# Hidden States
states = ["Walking", "Running", "Resting"]

# Observation Symbols
observations = ["Low", "Medium", "High"]

# Initial Probabilities
pi = [0.5, 0.3, 0.2]

# Transition Matrix
A = [
    [0.6, 0.3, 0.1],
    [0.2, 0.5, 0.3],
    [0.3, 0.2, 0.5]
]

# Emission Matrix
B = [
    [0.6, 0.3, 0.1],   # Walking
    [0.1, 0.3, 0.6],   # Running
    [0.7, 0.2, 0.1]    # Resting
]

# Observation Sequence
O = ["High", "Medium", "Low", "High"]

# ==========================================================
# CONVERT OBSERVATIONS TO INDEXES
# ==========================================================

obs_index = []

for obs in O:
    for i in range(len(observations)):
        if observations[i] == obs:
            obs_index.append(i)

# ==========================================================
# FORWARD TABLE
# ==========================================================

T = len(O)
N = len(states)

alpha = []

for i in range(T):
    row = []
    for j in range(N):
        row.append(0.0)
    alpha.append(row)

# ==========================================================
# INITIALIZATION
# α1(i) = π(i) * b(i,O1)
# ==========================================================

for i in range(N):
    alpha[0][i] = pi[i] * B[i][obs_index[0]]

# ==========================================================
# RECURSION
# αt(j)=Σ α(t-1)(i)a(i,j) * b(j,Ot)
# ==========================================================

for t in range(1, T):

    for j in range(N):

        total = 0

        for i in range(N):

            total = total + alpha[t-1][i] * A[i][j]

        alpha[t][j] = total * B[j][obs_index[t]]

# ==========================================================
# DISPLAY FORWARD TABLE
# ==========================================================

print("\nFORWARD PROBABILITY TABLE\n")

print("Time\tWalking\t\tRunning\t\tResting")

for t in range(T):

    print(
        t+1,
        "\t",
        round(alpha[t][0], 6),
        "\t",
        round(alpha[t][1], 6),
        "\t",
        round(alpha[t][2], 6)
    )

# ==========================================================
# TERMINATION
# ==========================================================

likelihood = 0

for i in range(N):
    likelihood = likelihood + alpha[T-1][i]

print("\nLikelihood P(O) =", likelihood)
