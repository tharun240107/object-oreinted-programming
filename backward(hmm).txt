# ==========================================================
# VITERBI ALGORITHM FOR HMM
# ==========================================================

states = ["Walking", "Running", "Resting"]

observations = ["Low", "Medium", "High"]

pi = [0.5, 0.3, 0.2]

A = [
    [0.6, 0.3, 0.1],
    [0.2, 0.5, 0.3],
    [0.3, 0.2, 0.5]
]

B = [
    [0.6, 0.3, 0.1],
    [0.1, 0.3, 0.6],
    [0.7, 0.2, 0.1]
]

O = ["High", "Medium", "Low", "High"]

# ==========================================================
# OBSERVATION INDEX
# ==========================================================

obs_index = []

for obs in O:
    for i in range(len(observations)):
        if observations[i] == obs:
            obs_index.append(i)

T = len(O)
N = len(states)

# ==========================================================
# DELTA TABLE
# ==========================================================

delta = []

for t in range(T):
    row = []
    for j in range(N):
        row.append(0.0)
    delta.append(row)

# ==========================================================
# PSI TABLE
# ==========================================================

psi = []

for t in range(T):
    row = []
    for j in range(N):
        row.append(0)
    psi.append(row)

# ==========================================================
# INITIALIZATION
# ==========================================================

for i in range(N):

    delta[0][i] = (
        pi[i]
        * B[i][obs_index[0]]
    )

    psi[0][i] = 0

# ==========================================================
# RECURSION
# ==========================================================

for t in range(1, T):

    for j in range(N):

        max_value = -1
        max_state = 0

        for i in range(N):

            value = (
                delta[t-1][i]
                * A[i][j]
            )

            if value > max_value:

                max_value = value
                max_state = i

        delta[t][j] = (
            max_value
            * B[j][obs_index[t]]
        )

        psi[t][j] = max_state

# ==========================================================
# DISPLAY DELTA TABLE
# ==========================================================

print("\nVITERBI TABLE\n")

print("Time\tWalking\t\tRunning\t\tResting")

for t in range(T):

    print(
        t+1,
        "\t",
        round(delta[t][0],6),
        "\t",
        round(delta[t][1],6),
        "\t",
        round(delta[t][2],6)
    )

# ==========================================================
# TERMINATION
# ==========================================================

best_prob = -1
last_state = 0

for i in range(N):

    if delta[T-1][i] > best_prob:

        best_prob = delta[T-1][i]
        last_state = i

# ==========================================================
# BACKTRACKING
# ==========================================================

path = []

for i in range(T):
    path.append(0)

path[T-1] = last_state

for t in range(T-2, -1, -1):

    path[t] = psi[t+1][path[t+1]]

# ==========================================================
# OUTPUT
# ==========================================================

print("\nMost Likely Hidden State Sequence:\n")

for i in range(T):
    print(states[path[i]], end=" ")

print("\n\nProbability =", best_prob)
